def getDataDependenceSum(n):
    total_sum = 0
    
    # Start from x = 0 up to n
    x = 0
    while x <= n:
        # Calculate the maximum k for the current x
        if x == 0:
            k_max = n + 1  # For x = 0, k can be any value greater than n
        else:
            k_max = n // x
            
        # Calculate the minimum k for the current x
        if x == 0:
            k_min = n + 1  # For x = 0, there's no valid k
        else:
            k_min = n // (x + 1) + 1  # k must be greater than n/(x + 1)
        
        # If k_min <= k_max, it means we have valid k values for this x
        if k_min <= k_max:
            total_sum += x
        
        x += 1
    
    return total_sum

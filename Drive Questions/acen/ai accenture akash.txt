# Read only region start
class UserMainCode(object):
    @classmethod
    def fourVariables(cls, input1):
        # Read only region end
        # Write code here

        left, right = input1.split('=')
        a, b = left.split('+')
        c, d = right.split('+')

        operands = [a, b, c, d]   

        left_sum  = sum(int(op) for op in [a, b] if op.isdigit())
        right_sum = sum(int(op) for op in [c, d] if op.isdigit())

        result = []
        for i, op in enumerate(operands):
            if op.isdigit():
                result.append(int(op))
            else:
                if i < 2:
                    result.append(right_sum - left_sum)
                else:
                    result.append(left_sum - right_sum)
        result.sort()
        return ' '.join(map(str, result))


# Test case 1
print(UserMainCode.fourVariables("2+4=c+5"))   # Expected: 1 2 4 5

# Test case 2
print(UserMainCode.fourVariables("a+3=5+6"))   # Expected: 3 5 6 8
